import streamlit as st
import pydeck as pdk
import pandas as pd

# ---------------------------------------------------------
# 1. Dashboard Configuration
# ---------------------------------------------------------
st.set_page_config(page_title="Title 18 3D Massing & Parking Tool", layout="wide")
st.title("Interactive 3D Massing: Title 18 vs. Standard IBC")
st.markdown("Toggle the regulatory pathways to visualize building dimensions, reclaimed unit space, and off-street parking allocations on a standard infill parcel.")

# ---------------------------------------------------------
# 2. Define Spatial Data (Coordinates for a conceptual parcel)
# ---------------------------------------------------------
# Base coordinates (Using an El Paso baseline longitude/latitude)
BASE_LON, BASE_LAT = -106.4850, 31.7619

# Helper to generate polygon offsets
def offset(lon_off, lat_off):
    return [BASE_LON + lon_off, BASE_LAT + lat_off]

# ---------------------------------------------------------
# 3. Model Geometries & Extrusions
# ---------------------------------------------------------
# IBC Baseline: Sprawling footprint, 2 stair cores, large utility vault
ibc_data = [
    {"component": "Apartment Units", "color": [100, 150, 250, 200], "height": 10, "polygon": [[offset(0,0), offset(0.0004,0), offset(0.0004, 0.00015), offset(0, 0.00015)]]},
    {"component": "Stair Tower 1", "color": [250, 100, 100, 255], "height": 12, "polygon": [[offset(-0.00005, 0), offset(0,0), offset(0, 0.00015), offset(-0.00005, 0.00015)]]},
    {"component": "Stair Tower 2", "color": [250, 100, 100, 255], "height": 12, "polygon": [[offset(0.0004, 0), offset(0.00045,0), offset(0.00045, 0.00015), offset(0.0004, 0.00015)]]},
    {"component": "Commercial Utility Vault", "color": [150, 150, 150, 255], "height": 2, "polygon": [[offset(0.00015, -0.00005), offset(0.00025,-0.00005), offset(0.00025, 0), offset(0.00015, 0)]]},
    {"component": "Off-Street Parking", "color": [50, 50, 50, 150], "height": 0.1, "polygon": [[offset(0, 0.0002), offset(0.0004, 0.0002), offset(0.0004, 0.0003), offset(0, 0.0003)]]},
]

# Title 18: Compact footprint, 1 central stair, no vault, reclaimed space
title18_data = [
    {"component": "Apartment Units", "color": [100, 150, 250, 200], "height": 10, "polygon": [[offset(0,0), offset(0.00015,0), offset(0.00015, 0.00015), offset(0, 0.00015)]]},
    {"component": "Reclaimed Living Space", "color": [100, 250, 150, 255], "height": 10, "polygon": [[offset(0.0002, 0), offset(0.00035,0), offset(0.00035, 0.00015), offset(0.0002, 0.00015)]]},
    {"component": "Central Stair Core", "color": [250, 100, 100, 255], "height": 12, "polygon": [[offset(0.00015, 0), offset(0.0002,0), offset(0.0002, 0.00015), offset(0.00015, 0.00015)]]},
    {"component": "Expanded Off-Street Parking / Courtyard", "color": [50, 50, 50, 150], "height": 0.1, "polygon": [[offset(0, 0.00018), offset(0.00035, 0.00018), offset(0.00035, 0.00035), offset(0, 0.00035)]]},
]

# ---------------------------------------------------------
# 4. Sidebar Controls & Logic
# ---------------------------------------------------------
st.sidebar.header("Scenario Controls")
scenario = st.sidebar.radio("Select Code Pathway:", ("Standard IBC (Dual Stair)", "Title 18 (Single Stair)"))

# Dynamic metric updates based on scenario
if scenario == "Standard IBC (Dual Stair)":
    active_data = ibc_data
    sqft_rentable = "6,000 sq ft"
    sqft_dead = "1,500 sq ft"
    parking_spaces = "6 Spaces (Constrained)"
else:
    active_data = title18_data
    sqft_rentable = "7,200 sq ft (+20%)"
    sqft_dead = "300 sq ft"
    parking_spaces = "9 Spaces (Expanded Rear)"

st.sidebar.divider()
st.sidebar.metric("Rentable Area", sqft_rentable)
st.sidebar.metric("Non-Rentable/Egress Area", sqft_dead)
st.sidebar.metric("Off-Street Parking Capacity", parking_spaces)

# ---------------------------------------------------------
# 5. PyDeck Layer Rendering
# ---------------------------------------------------------
df = pd.DataFrame(active_data)

layer = pdk.Layer(
    "PolygonLayer",
    df,
    get_polygon="polygon",
    get_fill_color="color",
    get_elevation="height",
    extruded=True,
    wireframe=True,
    get_line_color=[255, 255, 255, 100],
    pickable=True,
    auto_highlight=True,
)

view_state = pdk.ViewState(
    longitude=BASE_LON + 0.0002,
    latitude=BASE_LAT + 0.0001,
    zoom=18.5,
    pitch=45,
    bearing=25,
)

# Render map in Streamlit
st.pydeck_chart(pdk.Deck(
    layers=[layer],
    initial_view_state=view_state,
    tooltip={"text": "{component}\nHeight: {height}m"}
))
